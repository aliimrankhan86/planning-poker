# Live estimation rooms are per-session and hold no indexable content.
# Two Team Room lines, not one: "/t/" does not match "/ja/t/my-team", and a
# Japanese facilitator's invite link carries the locale prefix. "/*?room="
# needs no locale twin — the wildcard already covers "/ja/?room=ABC12".
Disallow: /t/
Disallow: /*/t/
Disallow: /*?room=

# Owner-only usage dashboard. Not prerendered, so the host falls back to the
# home document and a crawler that does not run JavaScript is told to index it.
# vercel.json sends X-Robots-Tag: noindex on the same path.
Disallow: /admin

# Every AI crawler is deliberately allowed by the wildcard above — GPTBot,
# ClaudeBot, PerplexityBot, OAI-SearchBot, Google-Extended and the rest. Being
# quotable in an answer engine is the point. Naming them in their own groups
# would replace, not extend, the wildcard rule, so the three Disallow lines
# above would stop applying to them.
